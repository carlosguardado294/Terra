cmake_minimum_required(VERSION 3.16)
project(procedural-world LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(procedural-world src/main.cpp src/WorldGenerator.cpp src/WorldRenderer.cpp src/WorldSave.cpp)
if(EMSCRIPTEN)
  include(FetchContent)
  set(PLATFORM Web CACHE STRING "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
  set_target_properties(procedural-world PROPERTIES OUTPUT_NAME index SUFFIX ".html")
  target_link_options(procedural-world PRIVATE -sUSE_GLFW=3 -sALLOW_MEMORY_GROWTH=1 -sFORCE_FILESYSTEM=1 -sSTACK_SIZE=1048576 -lidbfs.js "SHELL:--shell-file ${CMAKE_SOURCE_DIR}/web/shell.html")
else()
  find_package(raylib CONFIG REQUIRED)
endif()
target_link_libraries(procedural-world PRIVATE raylib)
